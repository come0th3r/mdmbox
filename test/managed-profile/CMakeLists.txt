cmake_minimum_required(VERSION 3.16)
project(mdmbox_managed_profile_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(yaml-cpp CONFIG REQUIRED)
add_executable(mdmbox_managed_profile_tests main.cpp ../../sub/ManagedProfile.cpp ../../sub/ManagedProfileStore.cpp ../../sub/MihomoValidator.cpp ../../sub/SubscriptionContract.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/InstallationIdentity.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/SubscriptionService.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/ManagedRuntimePolicy.cpp)
target_link_libraries(mdmbox_managed_profile_tests PRIVATE Qt6::Core yaml-cpp)
enable_testing()
add_test(NAME managed-profile COMMAND mdmbox_managed_profile_tests)
find_package(Qt6 REQUIRED COMPONENTS Network)
add_executable(mdmbox_managed_runtime_smoke runtime.cpp ../../sub/ManagedProfile.cpp ../../sub/ManagedRuntimePolicy.cpp ../../sub/MihomoValidator.cpp ../../sub/ManagedCoreProcess.cpp)
target_link_libraries(mdmbox_managed_runtime_smoke PRIVATE Qt6::Core Qt6::Network yaml-cpp)
target_sources(mdmbox_managed_runtime_smoke PRIVATE ../../sub/ManagedControllerProbe.cpp)
add_executable(mdmbox_controller_probe_tests controller.cpp ../../sub/ManagedControllerProbe.cpp)
target_link_libraries(mdmbox_controller_probe_tests PRIVATE Qt6::Core Qt6::Network)
add_test(NAME controller-probe COMMAND mdmbox_controller_probe_tests)
# Explicit binary argument required; ordinary CTest does not start a runtime.
if (WIN32)
    add_executable(mdmbox_console_signal ../../sys/managed-console-signal.cpp)
    add_dependencies(mdmbox_managed_runtime_smoke mdmbox_console_signal)
    add_test(NAME console-signal-invalid COMMAND mdmbox_console_signal 0)
    set_tests_properties(console-signal-invalid PROPERTIES WILL_FAIL TRUE)
endif ()
