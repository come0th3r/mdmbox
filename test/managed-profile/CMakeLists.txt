cmake_minimum_required(VERSION 3.16)
project(mdmbox_managed_profile_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(yaml-cpp CONFIG REQUIRED)
add_executable(mdmbox_managed_profile_tests main.cpp ../../sub/ManagedProfile.cpp ../../sub/ManagedProfileStore.cpp ../../sub/MihomoValidator.cpp ../../sub/SubscriptionContract.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/InstallationIdentity.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/SubscriptionService.cpp)
target_sources(mdmbox_managed_profile_tests PRIVATE ../../sub/ManagedRuntimePolicy.cpp)
target_link_libraries(mdmbox_managed_profile_tests PRIVATE Qt6::Core yaml-cpp)
enable_testing()
add_test(NAME managed-profile COMMAND mdmbox_managed_profile_tests)
find_package(Qt6 REQUIRED COMPONENTS Network)
add_executable(mdmbox_managed_runtime_smoke runtime.cpp ../../sub/ManagedProfile.cpp ../../sub/ManagedRuntimePolicy.cpp ../../sub/MihomoValidator.cpp ../../sub/ManagedCoreProcess.cpp)
target_link_libraries(mdmbox_managed_runtime_smoke PRIVATE Qt6::Core Qt6::Network yaml-cpp)
target_sources(mdmbox_managed_runtime_smoke PRIVATE ../../sub/ManagedControllerProbe.cpp)
target_sources(mdmbox_managed_runtime_smoke PRIVATE ../../sub/CoreOwnership.cpp)
target_sources(mdmbox_managed_runtime_smoke PRIVATE ../../sub/ManagedSession.cpp ../../sub/ManagedProfileStore.cpp ../../sub/SubscriptionContract.cpp)
set_target_properties(mdmbox_managed_runtime_smoke PROPERTIES AUTOMOC ON)
target_sources(mdmbox_managed_runtime_smoke PRIVATE ../../sub/ManagedSessionController.cpp ../../sub/ManagedSessionController.hpp)
add_executable(mdmbox_controller_probe_tests controller.cpp ../../sub/ManagedControllerProbe.cpp)
target_link_libraries(mdmbox_controller_probe_tests PRIVATE Qt6::Core Qt6::Network)
add_test(NAME controller-probe COMMAND mdmbox_controller_probe_tests)
add_executable(mdmbox_core_ownership_tests core-ownership.cpp ../../sub/CoreOwnership.cpp)
target_link_libraries(mdmbox_core_ownership_tests PRIVATE Qt6::Core)
add_test(NAME core-ownership COMMAND mdmbox_core_ownership_tests)
add_executable(mdmbox_process_cleanup_tests process-cleanup.cpp ../../sub/CoreOwnership.cpp)
target_link_libraries(mdmbox_process_cleanup_tests PRIVATE Qt6::Core)
add_test(NAME process-cleanup COMMAND mdmbox_process_cleanup_tests)
set_tests_properties(process-cleanup PROPERTIES TIMEOUT 25)
add_executable(mdmbox_cleanup_completion_tests cleanup-completion.cpp ../../sub/CoreOwnership.cpp)
target_link_libraries(mdmbox_cleanup_completion_tests PRIVATE Qt6::Core)
add_test(NAME cleanup-completion COMMAND mdmbox_cleanup_completion_tests)
set_tests_properties(cleanup-completion PROPERTIES TIMEOUT 20)
# Explicit binary argument required; ordinary CTest does not start a runtime.
