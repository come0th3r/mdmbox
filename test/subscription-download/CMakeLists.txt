cmake_minimum_required(VERSION 3.16)
project(mdmbox_subscription_download_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
add_executable(mdmbox_subscription_download_tests main.cpp ../../sub/SubscriptionDownloader.cpp ../../sub/SubscriptionContract.cpp)
target_compile_definitions(mdmbox_subscription_download_tests PRIVATE MDMBOX_DOWNLOADER_LOOPBACK_TEST)
target_link_libraries(mdmbox_subscription_download_tests PRIVATE Qt6::Core Qt6::Network)
enable_testing()
add_test(NAME subscription-download COMMAND mdmbox_subscription_download_tests)
add_executable(mdmbox_subscription_download_guard guard.cpp ../../sub/SubscriptionDownloader.cpp ../../sub/SubscriptionContract.cpp)
target_link_libraries(mdmbox_subscription_download_guard PRIVATE Qt6::Core Qt6::Network)
add_test(NAME subscription-production-guard COMMAND mdmbox_subscription_download_guard)
add_executable(mdmbox_subscription_tls_tests tls.cpp ../../sub/SubscriptionDownloader.cpp ../../sub/SubscriptionContract.cpp)
target_link_libraries(mdmbox_subscription_tls_tests PRIVATE Qt6::Core Qt6::Network)
# Certificates are ephemeral external fixtures; no HTTP exception in this target.
set(MDMBOX_TLS_TEST_CERT "" CACHE FILEPATH "Ephemeral localhost-only test certificate")
set(MDMBOX_TLS_TEST_KEY "" CACHE FILEPATH "Ephemeral test private key (never a real credential)")
if (MDMBOX_TLS_TEST_CERT AND MDMBOX_TLS_TEST_KEY)
    add_test(NAME subscription-tls COMMAND mdmbox_subscription_tls_tests "${MDMBOX_TLS_TEST_CERT}" "${MDMBOX_TLS_TEST_KEY}")
endif ()
